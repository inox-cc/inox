cmake_minimum_required(VERSION 3.20)

project(inox_cmake_example C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

find_program(NODE_EXECUTABLE node REQUIRED)

if(INOX_TLS_BACKEND STREQUAL "boringssl")
  enable_language(CXX)
endif()

set(INOX_CLI "../bin/inox.ts")
set(INOX_ENTRY "src/index.ts")
set(INOX_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")

add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../runtime/c" "${CMAKE_CURRENT_BINARY_DIR}/inox_runtime")

file(GLOB_RECURSE INOX_TS_SOURCES CONFIGURE_DEPENDS
  "src/*.ts"
)

file(GLOB_RECURSE INOX_COMPILER_SOURCES CONFIGURE_DEPENDS
  "../bin/*.ts"
  "../src/compiler/*.ts"
)

set(INOX_GENERATED_SOURCES)
set(INOX_GENERATED_HEADERS)
set(INOX_CLI_LOOP_ARGS)
set(INOX_CLI_TLS_ARGS)

if(DEFINED INOX_LOOP_BACKEND AND NOT INOX_LOOP_BACKEND STREQUAL "")
  list(APPEND INOX_CLI_LOOP_ARGS "--loop-backend" "${INOX_LOOP_BACKEND}")
endif()

if(DEFINED INOX_TLS_BACKEND AND NOT INOX_TLS_BACKEND STREQUAL "")
  list(APPEND INOX_CLI_TLS_ARGS "--tls-backend" "${INOX_TLS_BACKEND}")
endif()

foreach(INOX_TS_SOURCE IN LISTS INOX_TS_SOURCES)
  file(RELATIVE_PATH INOX_REL_TS "${CMAKE_CURRENT_SOURCE_DIR}" "${INOX_TS_SOURCE}")
  get_filename_component(INOX_REL_DIR "${INOX_REL_TS}" DIRECTORY)
  get_filename_component(INOX_NAME_WE "${INOX_REL_TS}" NAME_WE)

  if(INOX_REL_DIR STREQUAL "")
    list(APPEND INOX_GENERATED_SOURCES "${INOX_GENERATED_DIR}/${INOX_NAME_WE}.c")
    list(APPEND INOX_GENERATED_HEADERS "${INOX_GENERATED_DIR}/${INOX_NAME_WE}.h")
  else()
    list(APPEND INOX_GENERATED_SOURCES "${INOX_GENERATED_DIR}/${INOX_REL_DIR}/${INOX_NAME_WE}.c")
    list(APPEND INOX_GENERATED_HEADERS "${INOX_GENERATED_DIR}/${INOX_REL_DIR}/${INOX_NAME_WE}.h")
  endif()
endforeach()

add_custom_command(
  OUTPUT ${INOX_GENERATED_SOURCES}
  BYPRODUCTS ${INOX_GENERATED_HEADERS}
  COMMAND "${NODE_EXECUTABLE}" "${INOX_CLI}" "${INOX_ENTRY}" --emit c --out-dir "${INOX_GENERATED_DIR}" --entry ${INOX_CLI_LOOP_ARGS} ${INOX_CLI_TLS_ARGS}
  MAIN_DEPENDENCY "${INOX_ENTRY}"
  DEPENDS ${INOX_TS_SOURCES} ${INOX_COMPILER_SOURCES}
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  VERBATIM
)

add_executable(inox_cmake_example ${INOX_GENERATED_SOURCES})
target_link_libraries(inox_cmake_example PRIVATE inox_runtime)

if(INOX_TLS_BACKEND STREQUAL "boringssl")
  set_property(TARGET inox_cmake_example PROPERTY LINKER_LANGUAGE CXX)
endif()
