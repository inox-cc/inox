cmake_minimum_required(VERSION 3.20)

project(inox_cmake_example C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(INOX_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(INOX_COMPILER_MODE "node" CACHE STRING "inox compiler runner")
set_property(CACHE INOX_COMPILER_MODE PROPERTY STRINGS node native)
set(INOX_NODE_COMPILER "${INOX_REPO_ROOT}/compiler/index.ts" CACHE FILEPATH "inox hosted compiler")
set(INOX_NATIVE_COMPILER "${INOX_REPO_ROOT}/dist/inox" CACHE FILEPATH "inox native compiler")
set(INOX_ENTRY "src/index.ts")
set(INOX_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")

if(INOX_COMPILER_MODE STREQUAL "native")
  if(NOT EXISTS "${INOX_NATIVE_COMPILER}")
    message(FATAL_ERROR "dist/inox was not found. Run pnpm run build first.")
  endif()

  set(INOX_COMPILER_COMMAND "${INOX_NATIVE_COMPILER}")
  set(INOX_COMPILER_DEPENDENCY "${INOX_NATIVE_COMPILER}")
elseif(INOX_COMPILER_MODE STREQUAL "node")
  find_program(NODE_EXECUTABLE node REQUIRED)

  if(NOT EXISTS "${INOX_NODE_COMPILER}")
    message(FATAL_ERROR "compiler/index.ts was not found.")
  endif()

  set(INOX_COMPILER_COMMAND "${NODE_EXECUTABLE}" "${INOX_NODE_COMPILER}")
  set(INOX_COMPILER_DEPENDENCY "${INOX_NODE_COMPILER}")
else()
  message(FATAL_ERROR "INOX_COMPILER_MODE must be node or native.")
endif()

include("${INOX_REPO_ROOT}/cmake/InoxCompilerLibraries.cmake")
inox_prepare_compiler_library_native_plan()

add_subdirectory("${INOX_REPO_ROOT}/runtime" "${CMAKE_CURRENT_BINARY_DIR}/inox_runtime")

file(GLOB_RECURSE INOX_TS_SOURCES CONFIGURE_DEPENDS
  "src/*.ts"
)

file(GLOB_RECURSE INOX_COMPILER_SOURCES CONFIGURE_DEPENDS
  "${INOX_REPO_ROOT}/compiler/*.ts"
  "${INOX_REPO_ROOT}/stdlib/*.ts"
)

set(INOX_GENERATED_SOURCES)
set(INOX_GENERATED_HEADERS)
set(INOX_GENERATED_DECLARATIONS)
set(INOX_COMPILER_LOOP_ARGS)
set(INOX_COMPILER_TLS_ARGS)

if(DEFINED INOX_LOOP_BACKEND AND NOT INOX_LOOP_BACKEND STREQUAL "")
  list(APPEND INOX_COMPILER_LOOP_ARGS "--loop-backend" "${INOX_LOOP_BACKEND}")
endif()

if(DEFINED INOX_TLS_BACKEND AND NOT INOX_TLS_BACKEND STREQUAL "")
  list(APPEND INOX_COMPILER_TLS_ARGS "--tls-backend" "${INOX_TLS_BACKEND}")
endif()

foreach(INOX_TS_SOURCE IN LISTS INOX_TS_SOURCES)
  file(RELATIVE_PATH INOX_REL_TS "${CMAKE_CURRENT_SOURCE_DIR}" "${INOX_TS_SOURCE}")
  get_filename_component(INOX_REL_DIR "${INOX_REL_TS}" DIRECTORY)
  get_filename_component(INOX_NAME_WE "${INOX_REL_TS}" NAME_WE)

  if(INOX_REL_DIR STREQUAL "")
    list(APPEND INOX_GENERATED_SOURCES "${INOX_GENERATED_DIR}/${INOX_NAME_WE}.cc")
    list(APPEND INOX_GENERATED_HEADERS "${INOX_GENERATED_DIR}/${INOX_NAME_WE}.h")
    list(APPEND INOX_GENERATED_DECLARATIONS "${INOX_GENERATED_DIR}/${INOX_NAME_WE}.d.ts")
  else()
    list(APPEND INOX_GENERATED_SOURCES "${INOX_GENERATED_DIR}/${INOX_REL_DIR}/${INOX_NAME_WE}.cc")
    list(APPEND INOX_GENERATED_HEADERS "${INOX_GENERATED_DIR}/${INOX_REL_DIR}/${INOX_NAME_WE}.h")
    list(APPEND INOX_GENERATED_DECLARATIONS "${INOX_GENERATED_DIR}/${INOX_REL_DIR}/${INOX_NAME_WE}.d.ts")
  endif()
endforeach()

add_custom_command(
  OUTPUT ${INOX_GENERATED_SOURCES}
  BYPRODUCTS ${INOX_GENERATED_HEADERS} ${INOX_GENERATED_DECLARATIONS}
  COMMAND ${INOX_COMPILER_COMMAND} "${INOX_ENTRY}" --emit cc --out-dir "${INOX_GENERATED_DIR}" --entry ${INOX_COMPILER_LOOP_ARGS} ${INOX_COMPILER_TLS_ARGS}
  MAIN_DEPENDENCY "${INOX_ENTRY}"
  DEPENDS ${INOX_TS_SOURCES} ${INOX_COMPILER_SOURCES} "${INOX_COMPILER_DEPENDENCY}"
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  VERBATIM
)

add_executable(inox_cmake_example ${INOX_GENERATED_SOURCES})
target_link_libraries(inox_cmake_example PRIVATE inox_runtime)

set_property(TARGET inox_cmake_example PROPERTY LINKER_LANGUAGE CXX)
